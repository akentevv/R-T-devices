function [y, phase_acc, ftw, phase_index] = dds_core(Fclk, Fout_desired, Nacc, Nlut, Nsamp, amp_bits)
%dds_core Summary of this function goes here
%   Detailed explanation goes here

phase_mod = 2^Nacc;
lut_size = 2^Nlut;

% phase increment calculation 
ftw = round(Fout_desired * phase_mod / Fclk);

% Phase accimulator
n = 0:(Nsamp-1);
phase_acc = mod(n.*ftw, phase_mod);

% Phase quantization
phase_index = floor(phase_acc / 2^(Nacc - Nlut));

% sine LUT
lut = sin(2*pi* (0:(lut_size - 1)) / lut_size);

% Amplitude quantization

levels = 2^(amp_bits - 1) -1 ;

lut_q = round(lut * levels) / levels;

% Syntezed sin
y = lut_q(phase_index + 1);
end
